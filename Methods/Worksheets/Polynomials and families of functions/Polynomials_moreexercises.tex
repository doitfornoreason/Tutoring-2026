\section{Polynomials - more exercises}

\begin{enumerate}
    \item Let $p(x) = 3x^{4} + 4x^{3} + 6(n-2)x^{2} - 12n x + n^{2}$. 
    \begin{enumerate}
        \item Find values of $n$ s.t there is only one stationary point of $p$.
        \item If $p$ has only one stationary point, find values of $n$ for which $p(x) = 0$ has no solutions
    \end{enumerate}
    \item Find values of $p$ such that the function $f(x) = px^2+13x+p$ and $y = 12 x + 3p$ only has one intersection
    \item (CAS) State the set of values of $b$ s.t $2x^{3} + 3x^{2} - 5x + b = 0$ has 3 solutions
    \item Let $f'(x) = 2ax ^{3} + 3b x^{2} - 6 ax + b$ such that $f$ is convex on the interval $-1 < x< 1$ and concave elsewhere, and $f''(0) = 1$. Find possible values of $a,b$
    \item Let $f(x) = 3x^{3} + 2x^{2} + x$
    \begin{enumerate}
        \item Find the smallest angle between the tangent to the graph of $f$ at $x = 1$ and the x axis
        \item Find the acute angle between the line $y = -3x + 13$ and the tangent to $f$ at $x = \frac{1}{2}$
    \end{enumerate}
\end{enumerate}


\section{Differentiation}

\begin{enumerate}
    \item Find values of $p$ such that the absolute maximum occurs at $x = 10$
    \item Find values of $p$ such that the absolute minimum occurs at $x = 10$
\end{enumerate}

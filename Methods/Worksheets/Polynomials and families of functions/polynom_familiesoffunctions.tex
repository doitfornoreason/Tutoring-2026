\section{Transformations \& families of functions}

\begin{enumerate}
    \item The graph of $f(x) = \sqrt{x}$ undergoes the following transformations in order:
    \begin{enumerate}[i]
        \item dilation by a factor of 3 from the y axis
        \item reflection about the y axis
        \item translation of 6 units in the positive x direction
    \end{enumerate}
    Find the equation of the transformed function $g$, and state its domain and range
    \item Find a set of transformations that transforms the graph of $y = -\frac{3}{(2x - 1)^{2}} + 2$ to $y = \frac{1}{x^{2}}$
    \item State the transformations required to transfrom the graph of $y = \cos(x)$ to $y = -\frac{1}{5}\cos(3x - \frac{\pi}{2})$
    \item State the transformations required to transform the graph of $y = 3 \sin(\frac{1}{2}(x + \frac{\pi}{3})) - 5$ into the graph of $y = \sin(x)$
    \item For what values of $a$ is $y = ax^{2} - 3 a x + 1$ a perfect square?
    \item Find the turning points of the function $f(x) = \frac{4}{5}x^{5} - x^{4} - \frac{28}{3}x^{3} + 2x^{2} + 24x + 19$
    \item Let $f(x) = (x-1)^{2}(x -2) + 1$
    \begin{enumerate}
        \item Find the values for $p$ for which the equation $f(x) = p$ has exactly one solution
        \item Find the values for $h$ for which only one of the solutions to $f(x + h) + 1 = 0$ is positive
    \end{enumerate}

    \item Find a set of transformations that transforms the graph of $y = 2^{x}$ to $y = 3^{x}$
    \item Find values of p such that the function $f(x) = px^2+13x+p$ and $y = 12 x + 3p$ only has one intersection
\end{enumerate}

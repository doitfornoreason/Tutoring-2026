\noindent \textbf{Derivatives of inverse circular functions:}
\begin{align*}
    \frac{d}{dx}(\sin^{-1}(x)) &= \frac{1}{\sqrt{1-x^2}} \\[1ex]
    \frac{d}{dx}(\cos^{-1}(x)) &= \frac{-1}{\sqrt{1-x^2}} \\[1ex]
    \frac{d}{dx}(\tan^{-1}(x)) &= \frac{1}{1+x^2}
\end{align*}

\noindent \textbf{Points of inflection examples:}


\begin{itemize}
    \item $f''(x) = 0$ and there is
    \begin{enumerate}[1.]
        \item a point of inflection: $f(x) = x^3 + 3x^2 - x - 3$, $f(x) = \sin(x)$, $f(x) = \arctan(x)$, $f(x) = \dfrac{1}{\ln(x)}$.
        \item no point of inflection: $f(x) = a(x-h)^4 + k$.
    \end{enumerate}
    
    \item ($f''(x)$ is infinite or DNE and there is
    \begin{enumerate}[1.]
        \item a point of inflection: $f(x) = x|x|$, $f(x) = x^{1/3}$, $f(x) = x^{4/5} - x$.
        \item a change in concavity but no point of inflection: $f(x) = \dfrac{1}{x}$).
    \end{enumerate}
\end{itemize}

\vspace{2cm}

\noindent \textbf{Examples where the second derivative test:}
\begin{enumerate}[1.]
    \item gives a conclusion and is more efficient than the sign test: Any polynomial function.
    
    \vspace{0.5cm}
    
    \item gives a conclusion but is less efficient than the sign test: $f(x) = \dfrac{x-1}{x^2 - x + 1}$.
    
    \vspace{0.5cm}
    
    \item is inconclusive and the sign test must be used: $f(x) = (x-a)^3 + b$, $f(x) = (x-a)^4 + b$, $f(x) = x^4 \sin^2\left(\dfrac{1}{x}\right)$.
\end{enumerate}

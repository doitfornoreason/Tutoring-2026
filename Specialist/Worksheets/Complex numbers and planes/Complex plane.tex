\begin{itemize}
    \item \textbf{Lines}
    \begin{enumerate}[(a)]
        \item $a Re(z) + b Im(z) = c$
        \item $| z - z_{1} | = | z - z_{2} |$
    \end{enumerate}
    \begin{enumerate}
            \item Plot the line $\frac{| z - 3 |}{| z - 2 + i |} = 1$
            \vspace{2cm}
            \item Plot the region $3 Re(z - 2 + 3i) \leq 1 - Im(z - 1)$
            \vspace{2cm}
        \end{enumerate}
    \item \textbf{Rays}
    \begin{enumerate}[(a)]
        \item Defined via the arg function, usually of form $Arg(z-z_{1}) = \theta $
        \item Note $Arg(0)$ is not defined, so there is an open circle at the terminus
    \end{enumerate}
    \begin{enumerate}        
        \item Interpret what $a$ does in the graph of $Arg(az) = \frac{\pi}{4}$
        \vspace{2cm}
        \item Interpret what $b$ does in the graph of $Arg(z - b) = \frac{\pi}{4}$
        \vspace{2cm}
        \item Find the rule for the graph of $Arg(iz - 1) = \frac{\pi}{3}$ in Cartesian form
        \vspace{2cm}
        \item Find the rule for the graph of $Arg(z - 1) = Arg(z + 1)$ in Cartesian form
        \vspace{3cm}
    \end{enumerate}
    \item \textbf{Circles}
    \begin{enumerate}[(a)]
        \item $| z - z_{1} | = a$
        \item $(z - z_{1})(\bar{z} - \bar{z_{1}}) = a$ (This is similar to form (a), how?)
        \item $| z - z_{1} | = b | z - z_{2} |$, $b > 0, b \neq 1$
    \end{enumerate}
    \begin{enumerate}
        \item $4(2z + 3 - 2i)(2\bar{z} + 3 + 2i) = 25$
        \vspace{2cm}
        \item $|z - 2| = 2|z + 1|$
        \vspace{2cm}
    \end{enumerate}
\end{itemize}
